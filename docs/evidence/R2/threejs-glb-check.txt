// furniture-kit.ts logic, verbatim except for the optional dequantize step
function kitPositions(gltf, dequantize) {
  gltf.scene.updateMatrixWorld(true);
  const out = [];
  gltf.scene.traverse((o) => {
    if (!o.isMesh) return;
    const geo = o.geometry.clone();
    if (dequantize) for (const [name, a] of Object.entries(geo.attributes)) {
      if (a.array instanceof Float32Array && !a.normalized) continue;
      const f = new Float32Array(a.count * a.itemSize);
      for (let i = 0; i < a.count; i++) for (let k = 0; k < a.itemSize; k++) f[i * a.itemSize + k] = a.getComponent(i, k);
      geo.setAttribute(name, new THREE.BufferAttribute(f, a.itemSize));
    }
    geo.applyMatrix4(o.matrixWorld);
    const src = geo.index ? geo.toNonIndexed() : geo;
    for (const name of ['position', 'normal']) {
      const a = src.attributes[name];
      const sliced = new THREE.BufferAttribute(a.array.slice(0), a.itemSize);
      for (let i = 0; i < sliced.count; i++) out.push([name, sliced.getX(i), sliced.getY(i), sliced.getZ(i)]);
    }
  });
  return out;
}
