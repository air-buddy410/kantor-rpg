        pdf = PDF / "A-103.pdf"
        idx = opening_schedule_page(pdf)
        runs, s = text_runs(pdf, idx)
        ids = {d["id"] for d in wd["doors"]} | {w["id"] for w in wd["windows"]}
        rows = [r for r in runs if r[0] in ids]
